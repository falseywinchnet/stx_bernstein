cmake_minimum_required(VERSION 3.21)
project(stx_bernstein VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)
option(STX_BERNSTEIN_SANITIZE "Instrument the library and deterministic tests" OFF)
option(BUILD_SHARED_LIBS "Build a shared library (also used by the Python demonstrations)" OFF)
add_library(stx_bernstein src/bernstein.cpp)
add_library(stx::bernstein ALIAS stx_bernstein)
set_target_properties(stx_bernstein PROPERTIES EXPORT_NAME bernstein POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(stx_bernstein PUBLIC cxx_std_20)
target_include_directories(stx_bernstein PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(MSVC)
    target_compile_options(stx_bernstein PRIVATE /W4 /fp:strict)
else()
    target_compile_options(stx_bernstein PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
if(STX_BERNSTEIN_SANITIZE)
    if(MSVC)
        message(FATAL_ERROR "This sanitizer configuration requires Clang or GCC")
    endif()
    target_compile_options(stx_bernstein PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(stx_bernstein PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
    add_executable(bernstein_checks tests/checks.cpp tests/reference/surface_fit.cpp)
    target_link_libraries(bernstein_checks PRIVATE stx::bernstein)
    add_test(NAME bernstein_checks COMMAND bernstein_checks)
endif()
install(TARGETS stx_bernstein EXPORT stx_bernsteinTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/stx_bernstein.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/stx_bernstein)
install(EXPORT stx_bernsteinTargets NAMESPACE stx:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_bernstein)
configure_package_config_file(cmake/stx_bernsteinConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/stx_bernsteinConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_bernstein)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/stx_bernsteinConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/stx_bernsteinConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/stx_bernsteinConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_bernstein)
